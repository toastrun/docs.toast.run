hreflang links of the language tree A target that does not exist falls back to the front page of the target tree. {%- endcomment -%} {%- assign languages = site.data.languages -%} {%- assign versions elsif all_urls contains landing -%} {%- assign target = landing -%} {%- endif -%} {%- if target -%} {%- assign has_version_link = true -%} {{ version.label }} {%- endif -%} {%- endif -%} {%- endfor -%} {%- endcapture -%} {%- unless has_version_link -%} {%- assign version_menu_html = "" -%} {%- endunless -%} {%- endif -%}
